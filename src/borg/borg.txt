# Este es el archivo de configuración para el Borg. Para una visión general de alto nivel
# del Borg y sus características, consulta la documentación oficial en
# docs/hacking/borg.rst.

# Este archivo te permitirá personalizar tu borg según ciertos temas.
# El borg tiene bastante éxito, habiendo ganado el juego varias veces.
# Sin embargo, algunos jugadores/observadores de borgs quisieran verlo funcionar
# de manera diferente. A algunos les gustaría ver al borg jugar más agresivamente, o
# más conservadoramente. Mientras que otros quisieran valorar más la clase de armadura.
# Otros quisieran verlo recolectar más objetos de velocidad. Con las
# expresiones a continuación, puedes influir en el comportamiento del borg y
# en sus elecciones de selección de equipo. Para modificar, simplemente establece los valores en TRUE o FALSE,
# con la excepción de algunos números enteros.


### Adoraciones ###

# Las variables 'borg_worship_' influirán en su selección de equipo al
# añadir bonificaciones adicionales a los objetos que mejoran ese atributo. Por ejemplo:
# si un borg recibe 2000 pts por un solo punto de clase de armadura, un
# borg que 'adora la CA' recibiría 3000 pts. Esta bonificación influiría significativamente
# en las elecciones de equipo.
#
# borg_worships_gold explicado:
# La opción borg_worships_gold impactará enormemente en cómo el borg trata los objetos
# que encuentra en la mazmorra. Con esta opción activada, regresará a la ciudad
# cada vez que llene su inventario con objetos y luego venderá esos objetos. Esto puede causar
# que el borg pierda algunos artefactos, ya que los venderá incluso si no están identificados.
# En términos generales, el borg no venderá un objeto a menos que esté al menos
# pseudoidentificado. El beneficio de esta opción es que el borg aumentará significativamente
# sus ingresos en los niveles bajos. Esto aumentará su capacidad de supervivencia
# al permitirle invertir en armadura y flechas. Con esto activado,
# no recordará ir a la ciudad para vender cosas, sino que subirá las escaleras en su lugar.
# Tampoco recordará volver a la mazmorra hasta el nivel 8 (en lugar del 5).
# Esta opción no tiene efecto después del nivel de personaje 20. Otro inconveniente de esto es
# que el borg pasará más tiempo en la ciudad y puede tener que evadir más a los Veteranos.

borg_worships_damage = FALSE
borg_worships_speed = FALSE
borg_worships_hp = FALSE
borg_worships_mana = FALSE
borg_worships_ac = FALSE
borg_worships_gold = FALSE


### Arriesgado ###

# Un borg arriesgado es aquel que no está limitado por los requisitos de nivel de personaje
# para descender a más profundidad. También es más probable que permanezca en una batalla
# en lugar de teletransportarse para salir. Un borg arriesgado descenderá más rápido pero tiene más probabilidades de morir.
# A un borg arriesgado no le preocupa matar a los únicos antes de descender a más profundidad.
#
# Un borg que busca únicos activará la bandera auto_scum para generar
# niveles más emocionantes y con suerte encontrar un único
# 
# Los borgs que matan_unicos esperarán a que los únicos mueran antes de descender a más profundidad
# en la mazmorra. Por lo general, no descenderá más allá de tres únicos.
# Ejemplo: los Monstruos Únicos que normalmente residen en los niveles 12, 15 y 18 están
# todavía vivos, por lo que el borg no descenderá más allá del nivel 18 hasta que al menos uno
# de esos únicos esté muerto. Establecer esto en FALSE permitirá al borg descender
# más profundo, más rápido, sin ser retenido por los únicos. Pero corre el riesgo
# de encontrar 3 o 4 únicos en un solo nivel. Eso puede ser un poco aterrador.

borg_plays_risky = FALSE
borg_kills_uniques = FALSE


### Objetos de Intercambio ###

# El borg está diseñado para usar Objetos de Intercambio en el inventario. Un objeto de intercambio
# es un objeto de equipo (como una espada o una pieza de armadura) que posee una
# resistencia o rasgo deseable. Si el borg está en peligro, entonces
# considerará intercambiar su objeto actual por el objeto de intercambio. Un ejemplo principal de
# esto es en el nivel de mazmorra 60. Se requiere que tenga varias resistencias
# activas. Si no las tiene, no descenderá más profundo. Sin embargo,
# con los Intercambios activados, seleccionará un objeto para cubrir la resistencia abierta y luego
# descenderá. Tener Intercambios activados permite al borg descender más profundo, más rápido.
# La desventaja es que los Objetos de Intercambio ocupan dos espacios de inventario.

borg_uses_swaps = TRUE


### Opciones Extrañas ###

# algunas opciones de nacimiento no permiten que el borg funcione bien. Si quieres permitir
# que el borg funcione de todos modos, establece esto en TRUE y ejecuta bajo tu propio riesgo.

borg_allow_strange_opts = FALSE


### Engañar a la Muerte ###

# Si borg_cheat_death está activado, la opción para engañar a la muerte se activará y, cuando
# el borg muera, engañará a la muerte y continuará. Si esto es falso y la
# opción para engañar a la muerte en el juego es verdadera, el borg se reencarnará (reaparecerá)
# como un jugador usando las opciones a continuación para elegir al nuevo jugador. Si la opción
# para engañar a la muerte es falsa y esto es falso, el borg se detendrá al morir.

borg_cheat_death = FALSE


### Reaparición ###

# Si el borg está en modo de juego continuo (como con el protector de pantalla),
# reaparecerá un personaje generado aleatoriamente al morir. Esta sección
# permite al usuario establecer cierta preferencia por el tipo de personaje generado.
# Puedes seleccionar la raza y/o clase y/o estadísticas mínimas para el Borg.
# También puedes seleccionar el nivel de personaje mínimo necesario antes de que se
# cree el volcado de personaje en el directorio del juego.
#
# Con respecto a la raza, el borg está programado para aceptar la variedad de
# razas definidas por el usuario que puedan estar en el archivo p_info.txt. Por lo tanto, los valores
# dados en la tabla a continuación son para las razas predeterminadas. Pero si eres lo
# suficientemente inteligente como para modificar el archivo p_info.txt, entonces eres lo suficientemente inteligente para deducir
# que la raza 0 no es necesariamente igual a Humano. Es solo la primera
# raza listada en tu archivo p_info.txt.
#
# RAZA: -1 Raza Aleatoria         CLASE: -1 Clase Aleatoria
#        0 Humano                          0 Guerrero
#        1 Semielfo                        1 Mago
#        2 Elfo                            2 Druida
#        3 Hobbit                           3 Sacerdote
#        4 Gnomo                           4 Nigromante
#        5 Enano                           5 Paladín
#        6 Semiorco                        6 Pícaro
#        7 Semitrol                         7 Montaraz
#        8 Dúnadan                          8 Guardia Negra
#        9 Alto Elfo
#       10 Kobold
#
# Con respecto a las estadísticas. Las estadísticas mínimas se tiran por el
# borg. CON será 16, la estadística secundaria (generalmente DEX) será 17,
# La estadística primaria será 17. Estos son los valores máximos que
# se pueden obtener (las bonificaciones raciales y de clase se añaden después).
# Si el borg no puede alcanzar las estadísticas después de 500,000 intentos, el juego
# simplemente te asignará una estadística aleatoria.
#
# Después de matar a Morgoth, generalmente, el borg se detendrá y te permitirá
# inspeccionar la victoria. Si estableces respawn_winners, después de
# matar a Morgoth, el borg generará un nuevo personaje. Se crea un archivo de mapa
# para que puedas ver cómo fue la batalla final.

borg_respawn_race = -1
borg_respawn_class = -1
borg_respawn_winners = FALSE

### Tolerancia a Cofres ###

# El borg puede buscar, desarmar y abrir cofres. Existe un riesgo
# inherente al tratar con cofres. El riesgo de activar la trampa frente al
# beneficio del botín dentro del cofre. El juego usa una fórmula sencilla
# para calcular el éxito de un cofre. Es:
# randint(100) < la habilidad del jugador menos la dificultad del cofre.
# Las dificultades van de 1 a 63.

borg_chest_fail_tolerance = 7


### Velocidad del Borg ###

Puedes ralentizar al borg de un par de maneras. Una forma es establecer el
factor de retardo del juego base en el menú de opciones. Esto ralentizará al borg
y los efectos visuales (como disparos de proyectiles, explosiones). Si usas el
protector de pantalla, necesitarás cargar angband.exe, cargar el archivo guardado, hacer las
modificaciones, luego guardar y salir. Otra forma de ralentizar al borg es
usar la siguiente variable. La fórmula es:
factor_de_retardo_juego_base^2 + borg_delay_factor^2.

borg_delay_factor = 0


### Acumulación de Dinero ###

# Esto comenzó más como una broma, pero tiene un propósito útil. Si te
# encuentras comprando en el BM y realmente quieres ese objeto y está justo
# fuera de tu presupuesto, entonces puedes establecer una cantidad de dinero en esta variable
# y el borg correrá por la ciudad matando monstruos y recolectando objetos para
# obtener efectivo. Una vez que haya alcanzado su objetivo monetario, se desconectará y dejará de moverse.
# El borg esperará en el nivel, para que aparezcan más habitantes, los matará,
# comprará comida cuando sea necesario. Pero no abandonará el nivel hasta que se alcance el objetivo.
# Para desactivar la acumulación de dinero, deja esta variable en 0 (cero).
# Esto también simulará borg_worships_gold y permitirá al borg vender
# objetos no identificados que serán dejados en la ciudad. El borg huirá de la ciudad
# de los Tipos Espeluznantes. Así que ten esto en cuenta con borgs por debajo del nivel 6. El borg
# no acumulará a menos que tenga una fuente de luz permanente (frasco, o algún
# artefacto brillante). El borg se detendrá si la Tienda General se queda sin comida para comprar.
# No se recomienda para borgs con Regeneración y sin Digestión
# Lenta, ya que gastarán todo su dinero en comida. Pero si tienes un
# borg con el hechizo Saciar el Hambre, pueden quedarse en la ciudad para siempre.
# NOTA: si un humano quiere que el borg acumule hasta una cantidad particular, pon
# borg_self_scum a FALSE o de lo contrario comprará su propio equipo con ese dinero.

borg_money_scum_amount = 0


### Acumulación del Borg ###

# Esto es un complemento de la Acumulación de Dinero de arriba. El borg acumulará en la ciudad si
# hay una poción de estadística en el BM. Si el borg encuentra un objeto en la ciudad que realmente
# le gustaría tener, entonces permanecerá en la ciudad hasta que haya reunido suficiente
# dinero para comprarlo. El borg se comportará como el borg_money_scum_amount de arriba.
# Si esto es TRUE, entonces se le permite al borg acumular para objetos que desea.
# Para prohibir este comportamiento (que puede llevar mucho tiempo) ponlo FALSE.
# NOTA: si un humano quiere que el borg acumule hasta una cantidad particular, pon
# borg_self_scum a FALSE o de lo contrario comprará su propio equipo con ese dinero.
# Además, el borg debe tener al menos nivel 10, con luz permanente,
# un 65% de probabilidad de evitar el carterista, y mucha comida.

borg_self_scum = TRUE


### Modo Lunal Borg ###

# El borg descenderá a lo profundo de la mazmorra, incluso mientras aún es de
# bajo nivel, saltando entre niveles para tomar escaleras descendentes cercanas.
# Una vez profundo, recolectará objetos y luego recordará volver a la ciudad. No recordará
# volver a bajar a la mazmorra, sino que caminará todo el camino de regreso. Si quieres
# una victoria rápida, esta no es la opción para ti. Si quieres que el borg tenga
# estadísticas maximizadas al principio a costa de turnos de juego, entonces pruébalo.
# La mejor manera de usar esto es tener un borg que pueda lanzar Saciar el Hambre.
# Este es un viaje de ida para el borg y tendrá hambre.
# El borg también debería tener una fuente de luz permanente (como el frasco). De nuevo,
# viaje de ida. No se detendrá a reabastecer su fuente de combustible. Sin embargo,
# si tiene comida o combustible, los usará cuando sea apropiado. Tener
# velocidad extra también ayudará al borg. A menudo bajará las escaleras y se
# encontrará justo al lado de un monstruo. A veces, ese monstruo puede
# golpear primero a un borg, matándolo instantáneamente. Incluso +3 de velocidad ayudaría
# a evitar este tipo de muerte instantánea.
#
# El Modo Auto-Lunal permitirá al borg usar la acumulación de escaleras para
# descender profundo en la mazmorra por sus propios objetivos. Por ejemplo, si el borg ha
# estado en la profundidad 99, y ahora no tiene más pociones de curación, solo
# estará preparado para la profundidad 45. Permanecerá en zonas poco profundas de la mazmorra,
# desde la profundidad 1-10 esperando a que las tiendas roten su inventario. Luego intentará
# comprar una poción de curación en las tiendas de la ciudad. Si no hay ninguna disponible,
# regresará a las profundidades poco profundas de la mazmorra y esperará a que las tiendas roten. Él
# repetirá este patrón hasta que consiga una poción de curación. Si se le permite
# al borg auto-lunal, entonces saltará las escaleras hacia abajo hasta una profundidad
# donde se sienta cómodo. En el caso anterior, hará esto hasta
# aproximadamente la profundidad 40.

borg_lunal_mode = FALSE
borg_self_lunal = FALSE

# Hay tres secciones
#	Power   - se usa para calcular el "poder" del borg
#	Depth   - se usa para determinar a qué profundidad se siente preparado el borg (no ir más profundo que)
#	Restock - se usa para determinar si el borg siente que necesita ir de compras
#				esto es un subconjunto de estar preparado, pero también se usa por sí solo.
#
# fórmula especial:
#	depth( n ) - la profundidad a comprobar. si la condición es verdadera, no bajar de esta profundidad.
#	range(desde, hasta) - bucle desde hasta hasta y añadir valor al poder cada vez
#	value( tipo, nombre ) - devuelve un valor dependiente de los tipos
#		lista de tipos:
#		ACTIVATION: devuelve el número de objetos que lleva puestos el borg que usan esa activación
#		TRAIT: valor de la lista borg.trait. Esto se puede volcar con ^zh
#		CONFIG: valor de la configuración leída de borg.txt
#		CLASS: verdadero/falso (1/0) si la clase actual del borg coincide con el nombre
#		RANGE: índice en el rango(desde, hasta) de arriba, devuelve -1 si no se da ningún rango
#				también se usa para comprobaciones de profundidad como la profundidad que se está comprobando.
#		tipo de object.txt con nombre que coincida con el nombre de ese archivo. Conteo de ese objeto que el borg lleva o tiene puesto.
#			ej: value(potion, Cure Critical Wounds) es el número de pociones de CCW
#				** vale la pena notar que value(trait, amt potion ccw) será una cantidad calculada de ccw, que podría incluir otra curación
#				tal como hechizos que curan esa cantidad, por lo que puede no tener el mismo valor que value(potion, Cure Critical Wounds)
#	reward: cálculo de cuánto aumento de poder por cada uno, si tienes rango, o solo por tenerlo, si no lo tienes
#	condition: si se calcula como cero, no obtienes la recompensa. Para rangos, se calcula una vez antes de que se procese el rango
#
# nota sobre los cálculos: algunas órdenes de operaciones se manejan, otras no. En caso de duda, los paréntesis son tus amigos.
